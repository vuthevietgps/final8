# MANUAL SSH COMMANDS - Copy từng lệnh để chạy trên server

# ==========================================
# BƯỚC 1: KIỂM tra TÌNH TRẠNG HIỆN TẠI
# ==========================================

# Kiểm tra containers đang chạy
docker ps

# Kiểm tra thư mục media trên host
ls -la ./media/

# Kiểm tra port đang listen
netstat -tulpn | grep :8088

# ==========================================  
# BƯỚC 2: KIỂM TRA CONTAINERS
# ==========================================

# Kiểm tra backend container
docker exec final8new-backend ls -la /app/uploads/media/

# Kiểm tra frontend container
docker exec final8new-frontend ls -la /var/www/media/

# Test backend API
docker exec final8new-backend wget -qO- http://localhost:3000/api/media | head -100

# Test backend health
docker exec final8new-backend wget -qO- http://localhost:3000/health

# ==========================================
# BƯỚC 3: KIỂM TRA NGINX CONFIG
# ==========================================

# Xem nginx config
docker exec final8new-frontend cat /etc/nginx/conf.d/default.conf | grep -A 10 -B 5 "location /media"

# Test nginx config
docker exec final8new-frontend nginx -t

# ==========================================
# BƯỚC 4: KIỂM TRA LOGS
# ==========================================

# Xem backend logs
docker logs final8new-backend --tail 20

# Xem frontend logs
docker logs final8new-frontend --tail 20

# ==========================================
# BƯỚC 5: FIX PROBLEMS
# ==========================================

# Tạo thư mục media nếu thiếu
mkdir -p ./media/2025/10
mkdir -p ./media/2025/11  
mkdir -p ./media/2025/12
chown -R 1000:1000 ./media/
chmod -R 755 ./media/

# Restart containers
docker compose -p final8local -f docker-compose.server.full.yml down
docker compose -p final8local -f docker-compose.server.full.yml up -d

# Đợi khởi động
sleep 15

# Reload nginx
docker exec final8new-frontend nginx -s reload

# ==========================================
# BƯỚC 6: TEST FINAL
# ==========================================

# Test từ bên ngoài (thay your-domain)
curl -I http://your-domain:8088/media/

# Test một file cụ thể (nếu có)
# curl -I http://your-domain:8088/media/2025/10/some-file.jpg

# Test API
curl http://your-domain:8088/api/media | head -50

# ==========================================
# BƯỚC 7: ALTERNATIVE FIX - Nếu vẫn lỗi
# ==========================================

# Option 1: Proxy media qua backend thay vì nginx
# Sửa nginx config để proxy /media/ về backend
docker exec final8new-frontend bash -c '
cat > /tmp/media-proxy.conf << EOF
    # Proxy media requests to backend instead of serving static
    location /media/ {
        proxy_pass http://backend:3000;
        proxy_http_version 1.1;
        proxy_set_header Host \$host;
        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto \$scheme;
    }
EOF
'

# Backup và replace nginx config
docker exec final8new-frontend cp /etc/nginx/conf.d/default.conf /etc/nginx/conf.d/default.conf.backup

# Apply new config với proxy (chỉ chạy nếu cần)
# docker exec final8new-frontend bash -c 'sed "/location \/media\/ {/,/}/c\\$(cat /tmp/media-proxy.conf)" /etc/nginx/conf.d/default.conf.backup > /etc/nginx/conf.d/default.conf'

# Test và reload
docker exec final8new-frontend nginx -t
docker exec final8new-frontend nginx -s reload

echo "=== HƯỚNG DẪN SỬ DỤNG ==="
echo "1. Copy từng command ở trên để chạy theo thứ tự"  
echo "2. Kiểm tra output của mỗi lệnh"
echo "3. Nếu có lỗi, report lại để fix tiếp"
